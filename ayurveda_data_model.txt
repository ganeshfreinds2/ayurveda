-- Enable pgvector extension
CREATE EXTENSION IF NOT EXISTS vector;

-- Create Users table
CREATE TABLE Users (
    user_id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL, -- Hashed password (bcrypt)
    preferences JSONB, -- e.g., {"preferred_categories": ["Diabetes", "Heart Health"]}
    health_conditions JSONB, -- e.g., {"conditions": ["diabetes", "high BP"]}
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create User_Problems table
CREATE TABLE User_Problems (
    problem_id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES Users(user_id) ON DELETE CASCADE,
    problem_name VARCHAR(255) NOT NULL, -- e.g., "diabetes"
    description TEXT, -- User-provided details
    embedding VECTOR(384), -- Vector embedding for semantic similarity (384 dims for common models like BERT)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create Remedies table
CREATE TABLE Remedies (
    remedy_id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    category VARCHAR(100) NOT NULL, -- e.g., "Diabetes", "Digestive Health"
    tags JSONB, -- e.g., ["bitter gourd", "blood sugar"]
    related_problems JSONB, -- e.g., ["diabetes", "high blood sugar"]
    embedding VECTOR(384), -- Vector embedding for semantic similarity
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create Videos table
CREATE TABLE Videos (
    video_id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    youtube_id VARCHAR(50) NOT NULL, -- YouTube video ID
    description TEXT,
    category VARCHAR(100), -- e.g., "Swadeshi Chikitsa"
    tags JSONB, -- e.g., ["ayurveda", "home remedies"]
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create Articles table
CREATE TABLE Articles (
    article_id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    tags JSONB, -- e.g., ["jaggery", "nutrition"]
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create User_Saved_Remedies table (junction table)
CREATE TABLE User_Saved_Remedies (
    user_id INTEGER REFERENCES Users(user_id) ON DELETE CASCADE,
    remedy_id INTEGER REFERENCES Remedies(remedy_id) ON DELETE CASCADE,
    saved_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, remedy_id)
);

-- Create User_Interactions parent table (partitioned by range on created_at)
CREATE TABLE User_Interactions (
    interaction_id SERIAL,
    user_id INTEGER REFERENCES Users(user_id) ON DELETE CASCADE,
    content_type VARCHAR(50) NOT NULL, -- "remedy", "video", "article"
    content_id INTEGER NOT NULL, -- ID of remedy, video, or article
    interaction_type VARCHAR(50) NOT NULL, -- "view", "like", "search"
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) PARTITION BY RANGE (created_at);

-- Create partitions for User_Interactions (e.g., monthly chunks for 2025)
CREATE TABLE User_Interactions_2025_01 PARTITION OF User_Interactions
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');
CREATE TABLE User_Interactions_2025_02 PARTITION OF User_Interactions
    FOR VALUES FROM ('2025-02-01') TO ('2025-03-01');
-- Add more partitions as needed

-- Add primary key to User_Interactions (on partitions)
ALTER TABLE User_Interactions_2025_01 ADD PRIMARY KEY (interaction_id);
ALTER TABLE User_Interactions_2025_02 ADD PRIMARY KEY (interaction_id);

-- Create Recommendations parent table (partitioned by range on created_at)
CREATE TABLE Recommendations (
    recommendation_id SERIAL,
    user_id INTEGER REFERENCES Users(user_id) ON DELETE CASCADE,
    remedy_id INTEGER REFERENCES Remedies(remedy_id) ON DELETE CASCADE,
    score DECIMAL(5, 2) NOT NULL, -- Recommendation score (0 to 1)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) PARTITION BY RANGE (created_at);

-- Create partitions for Recommendations (e.g., monthly chunks for 2025)
CREATE TABLE Recommendations_2025_01 PARTITION OF Recommendations
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');
CREATE TABLE Recommendations_2025_02 PARTITION OF Recommendations
    FOR VALUES FROM ('2025-02-01') TO ('2025-03-01');
-- Add more partitions as needed

-- Add primary key to Recommendations (on partitions)
ALTER TABLE Recommendations_2025_01 ADD PRIMARY KEY (recommendation_id);
ALTER TABLE Recommendations_2025_02 ADD PRIMARY KEY (recommendation_id);

-- Create Metadata table (for disclaimer)
CREATE TABLE Metadata (
    key VARCHAR(50) PRIMARY KEY,
    value TEXT
);

-- Create Indexes
-- Users
CREATE INDEX idx_users_email ON Users (email);

-- User_Problems
CREATE INDEX idx_user_problems_user_id ON User_Problems (user_id);
CREATE INDEX idx_user_problems_name ON User_Problems (problem_name);
CREATE INDEX idx_user_problems_embedding ON User_Problems USING hnsw (embedding vector_cosine_ops); -- Vector index for similarity

-- Remedies
CREATE INDEX idx_remedies_title ON Remedies (title);
CREATE INDEX idx_remedies_category ON Remedies (category);
CREATE INDEX idx_remedies_tags ON Remedies USING GIN (tags);
CREATE INDEX idx_remedies_related_problems ON Remedies USING GIN (related_problems);
CREATE INDEX idx_remedies_embedding ON Remedies USING hnsw (embedding vector_cosine_ops); -- Vector index for similarity

-- User_Interactions
CREATE INDEX idx_user_interactions_user_id ON User_Interactions (user_id);
CREATE INDEX idx_user_interactions_content ON User_Interactions (content_type, content_id);

-- Recommendations
CREATE INDEX idx_recommendations_user_id ON Recommendations (user_id);
