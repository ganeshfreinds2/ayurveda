// middleware/auth.js
const jwt = require('jsonwebtoken');
const dotenv = require('dotenv');

dotenv.config();

const auth = (req, res, next) => {
    const token = req.header('Authorization')?.replace('Bearer ', '');
    if (!token) return res.status(401).json({ error: 'No token provided' });

    try {
        const decoded = jwt.verify(token, process.env.JWT_SECRET);
        req.user = decoded;
        next();
    } catch (err) {
        res.status(401).json({ error: 'Invalid token' });
    }
};

module.exports = auth;

const express = require('express');

const jwt = require('jsonwebtoken');
const User = require('../models/User');

const router = express.Router();

// Register
router.post('/register', async (req, res) => {
    const { email, username, password } = req.body;
    try {
        const hashedPassword = await bcrypt.hash(password, 10);
        const user = await User.create({
            email,
            username,
            password_hash: hashedPassword,
        });
        const token = jwt.sign({ user_id: user.user_id }, process.env.JWT_SECRET, { expiresIn: '1h' });
        res.status(201).json({ token });
    } catch (err) {
        res.status(400).json({ error: err.message });
    }
});

// Login
router.post('/login', async (req, res) => {
    const { email, password } = req.body;
    try {
        const user = await User.findOne({ where: { email } });
        if (!user) return res.status(400).json({ error: 'Invalid credentials' });

        const isMatch = await bcrypt.compare(password, user.password_hash);
        if (!isMatch) return res.status(400).json({ error: 'Invalid credentials' });

        const token = jwt.sign({ user_id: user.user_id }, process.env.JWT_SECRET, { expiresIn: '1h' });
        res.json({ token });
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

// routes/recommendations.js
const express = require('express');
const Recommendation = require('../models/Recommendation');
const Remedy = require('../models/Remedy');
const UserProblem = require('../models/UserProblem');
const sequelize = require('../config/db');
const redisClient = require('../config/redis');
const auth = require('../middleware/auth');
const router = express.Router();

// Get recommendations
router.get('/:user_id', auth, async (req, res) => {
    const { user_id } = req.params;
    try {
        // Check Redis cache
        const cacheKey = `recommendations:user:${user_id}`;
        const cached = await redisClient.get(cacheKey);
        if (cached) return res.json(JSON.parse(cached));

        // Fetch from database
        const recommendations = await sequelize.query(`
            SELECT r.remedy_id, r.title, r.description, rec.score
            FROM Recommendations rec
            JOIN Remedies r ON rec.remedy_id = r.remedy_id
            WHERE rec.user_id = :user_id
            ORDER BY rec.score DESC
            LIMIT 10
        `, {
            replacements: { user_id },
            type: sequelize.QueryTypes.SELECT,
        });

        // Cache results
        await redisClient.setEx(cacheKey, 3600, JSON.stringify(recommendations));
        res.json(recommendations);
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

// Get real-time recommendations for a problem
router.get('/problem/:problem_id', auth, async (req, res) => {
    const { problem_id } = req.params;
    try {
        const problem = await UserProblem.findByPk(problem_id);
        if (!problem) return res.status(404).json({ error: 'Problem not found' });

        const recommendations = await sequelize.query(`
            SELECT r.remedy_id, r.title, r.description,
                   (r.embedding <=> :problem_embedding) AS distance
            FROM Remedies r
            WHERE r.related_problems @> :problem_name::jsonb
            ORDER BY distance ASC
            LIMIT 10
        `, {
            replacements: {
                problem_name: `["${problem.problem_name}"]`,
                problem_embedding: problem.embedding || '[0]'::VECTOR,
            },
            type: sequelize.QueryTypes.SELECT,
        });

        res.json(recommendations);
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

module.exports = router;

// services/embeddingService.js
const { PythonShell } = require('python-shell');

const generateEmbedding = async (text) => {
    if (!text) return null;
    return new Promise((resolve, reject) => {
        PythonShell.runString(`
            from sentence_transformers import SentenceTransformer
            model = SentenceTransformer('all-MiniLM-L6-v2')
            embedding = model.encode('${text}').tolist()
            print(embedding)
        `, null, (err, results) => {
            if (err) return reject(err);
            resolve(JSON.parse(results[0]));
        });
    });
};

module.exports = { generateEmbedding };

// jobs/recommendationJob.js
const cron = require('node-cron');
const sequelize = require('../config/db');
const User = require('../models/User');
const UserProblem = require('../models/UserProblem');
const Remedy = require('../models/Remedy');
const Recommendation = require('../models/Recommendation');
const redisClient = require('../config/redis');

const start = () => {
    // Run daily at midnight
    cron.schedule('0 0 * * *', async () => {
        console.log('Running recommendation job...');
        try {
            const users = await User.findAll();
            for (const user of users) {
                const problems = await UserProblem.findAll({ where: { user_id: user.user_id } });
                const remedies = await Remedy.findAll();

                const recommendations = [];
                for (const problem of problems) {
                    for (const remedy of remedies) {
                        let score = 0;

                        // Vector similarity (50%)
                        if (problem.embedding && remedy.embedding) {
                            const distance = await sequelize.query(
                                'SELECT (:problem_embedding <=> :remedy_embedding) AS distance',
                                {
                                    replacements: {
                                        problem_embedding: problem.embedding,
                                        remedy_embedding: remedy.embedding,
                                    },
                                    type: sequelize.QueryTypes.SELECT,
                                }
                            );
                            score += 0.5 * (1 - distance[0].distance);
                        }

                        // Content-based (30%)
                        if (remedy.related_problems?.includes(problem.problem_name)) {
                            score += 0.3;
                        }

                        // Preferences (20%)
                        if (user.preferences?.preferred_categories?.includes(remedy.category)) {
                            score += 0.2;
                        }

                        if (score > 0) {
                            recommendations.push({
                                user_id: user.user_id,
                                remedy_id: remedy.remedy_id,
                                score,
                            });
                        }
                    }
                }

                // Insert recommendations
                await Recommendation.destroy({ where: { user_id: user.user_id } });
                await Recommendation.bulkCreate(recommendations);

                // Cache recommendations
                const cacheKey = `recommendations:user:${user.user_id}`;
                const cachedRecommendations = await sequelize.query(`
                    SELECT r.remedy_id, r.title, r.description, rec.score
                    FROM Recommendations rec
                    JOIN Remedies r ON rec.remedy_id = r.remedy_id
                    WHERE rec.user_id = :user_id
                    ORDER BY rec.score DESC
                    LIMIT 10
                `, {
                    replacements: { user_id: user.user_id },
                    type: sequelize.QueryTypes.SELECT,
                });
                await redisClient.setEx(cacheKey, 3600 * 24, JSON.stringify(cachedRecommendations));
            }
            console.log('Recommendation job completed.');
        } catch (err) {
            console.error('Recommendation job error:', err);
        }
    });
};

module.exports = { start };
